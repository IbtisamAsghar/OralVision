// main.dart - OralVision Diagnostic UI Fragment
import 'package:flutter/material.dart';

void main() => runApp(OralVisionApp());

class OralVisionApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      theme: ThemeData(useMaterial3: true, colorSchemeSeed: Colors.blue),
      home: DiagnosticScreen(),
    );
  }
}

class DiagnosticScreen extends StatefulWidget {
  @override
  _DiagnosticScreenState createState() => _DiagnosticScreenState();
}

class _DiagnosticScreenState extends State<DiagnosticScreen> {
  bool _isAnalyzing = false;
  double _confidence = 0.0;

  void _analyzeImage() async {
    setState(() => _isAnalyzing = true);
    // Simulating API Call to FastAPI Backend
    await Future.delayed(Duration(seconds: 3));
    setState(() {
      _isAnalyzing = false;
      _confidence = 0.94; // 94% confidence
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text("OralVision AI Scan")),
      body: Column(
        children: [
          Expanded(
            child: Container(
              margin: EdgeInsets.all(20),
              decoration: BoxDecoration(
                color: Colors.grey[200],
                borderRadius: BorderRadius.circular(20),
                border: Border.all(color: Colors.blue, width: 2),
              ),
              child: Center(
                child: _isAnalyzing 
                  ? CircularProgressIndicator() 
                  : Icon(Icons.add_a_photo, size: 80, color: Colors.blue),
              ),
            ),
          ),
          if (_confidence > 0) 
            Padding(
              padding: EdgeInsets.all(16.0),
              child: Card(
                color: Colors.green[50],
                child: ListTile(
                  leading: Icon(Icons.check_circle, color: Colors.green),
                  title: Text("Detection: Dental Caries"),
                  subtitle: Text("Confidence Score: ${(_confidence * 100).toStringAsFixed(1)}%"),
                ),
              ),
            ),
          Padding(
            padding: EdgeInsets.symmetric(vertical: 20),
            child: ElevatedButton.icon(
              onPressed: _analyzeImage,
              icon: Icon(Icons.search),
              label: Text("Start AI Diagnosis"),
              style: ElevatedButton.styleFrom(padding: EdgeInsets.symmetric(horizontal: 40, vertical: 15)),
            ),
          )
        ],
      ),
    );
  }
}
